// =======================================================
// TELA 1: tela1-inicio.js (Página Inicial e Categorias)
// =======================================================

// 1. Variável em memória para guardar o nome do usuário
let nomeUsuario = "Visitante";

// 2. Variável de listagem com Array de objetos
let listaCategorias = [
  { Item: "Bebidas" },
  { Item: "Hortifruti" },
  { Item: "Limpeza" },
  { Item: "Padaria" }
];

// Selecionando elementos do HTML
const inputNome = document.getElementById("input-nome");
const btnSalvarNome = document.getElementById("btn-salvar-nome");
const mensagemBoasVindas = document.getElementById("mensagem-boas-vindas");

const inputPesquisaCat = document.getElementById("input-pesquisa-cat");
const inputCategoria = document.getElementById("input-categoria");
const btnAddCategoria = document.getElementById("btn-add-categoria");
const ulCategorias = document.getElementById("lista-categorias");

const btnDestaque = document.getElementById("btn-destaque");
const painelPrincipal = document.getElementById("painel-principal");

// Função para mostrar o array de categorias na tela (com filtro da barra de pesquisa)
function renderizarCategorias(filtro = "") {
  ulCategorias.innerHTML = "";

  listaCategorias
    .filter((cat) => cat.Item.toLowerCase().includes(filtro.toLowerCase()))
    .forEach((categoria) => {
      const li = document.createElement("li");
      li.textContent = categoria.Item;
      li.classList.add("tag-item"); // Adicionando classe CSS pelo JS

      // Ao clicar na categoria, adiciona/remove a classe de seleção
      li.addEventListener("click", () => {
        li.classList.toggle("tag-selecionada");
      });

      ulCategorias.appendChild(li);
    });
}

// Evento da barra de pesquisa em tempo real
inputPesquisaCat.addEventListener("input", () => {
  renderizarCategorias(inputPesquisaCat.value);
});

// Evento para salvar o nome digitado no input em memória
btnSalvarNome.addEventListener("click", () => {
  const valorDigitado = inputNome.value.trim();

  if (valorDigitado !== "") {
    nomeUsuario = valorDigitado; // Atualiza variável em memória
    mensagemBoasVindas.textContent = `Olá, ${nomeUsuario}! Clique nas categorias para destacá-las.`;
    // Adiciona classe de estilo do JS para o HTML
    mensagemBoasVindas.classList.add("texto-confirmado");
    inputNome.value = "";
  }
});

// Evento para adicionar nova categoria no Array em memória
btnAddCategoria.addEventListener("click", () => {
  const novaCategoria = inputCategoria.value.trim();

  if (novaCategoria !== "") {
    listaCategorias.push({ Item: novaCategoria });
    inputCategoria.value = "";
    inputPesquisaCat.value = "";
    renderizarCategorias();
  }
});

// Evento para alternar classe de estilo (.borda-neon) no painel principal
btnDestaque.addEventListener("click", () => {
  painelPrincipal.classList.toggle("borda-neon");
});

// Inicializa a listagem ao abrir a página
renderizarCategorias();

// =======================================================
// TELA 2: tela2-compras.js (Lista de Compras Interativa)
// =======================================================

// 1. Variável de listagem com Array (exatamente como solicitado no exemplo)
let listaCompras = [
  { Item: "Refrigerante", quantidade: 2, comprado: false },
  { Item: "Arroz 5kg", quantidade: 1, comprado: true },
  { Item: "Pão de Forma", quantidade: 1, comprado: false }
];

// Selecionando elementos do HTML
const inputPesquisaProduto = document.getElementById("input-pesquisa-produto");
const inputProduto = document.getElementById("input-produto");
const inputQtd = document.getElementById("input-qtd");
const btnAdicionar = document.getElementById("btn-adicionar");
const btnLimparComprados = document.getElementById("btn-limpar-comprados");
const ulCompras = document.getElementById("ul-compras");
const contadorItens = document.getElementById("contador-itens");

// Função que desenha a lista na tela com base no Array em memória e filtro de busca
function atualizarListaNaTela(termoBusca = "") {
  ulCompras.innerHTML = "";

  listaCompras.forEach((produto, index) => {
    // Se houver texto na barra de pesquisa, filtra os itens
    if (!produto.Item.toLowerCase().includes(termoBusca.toLowerCase())) {
      return;
    }

    const li = document.createElement("li");

    // Adiciona a classe base pelo JS
    li.classList.add("item-compra");

    // Se o item estiver marcado como comprado, adiciona a classe de estilo '.item-comprado'
    if (produto.comprado) {
      li.classList.add("item-comprado");
    }

    li.innerHTML = `
      <span>${produto.comprado ? "✅" : "⬜"} ${produto.Item}</span>
      <span class="badge rounded-pill text-bg-primary">${produto.quantidade} un.</span>
    `;

    // Ao clicar no item, alterna o estado em memória e a classe CSS
    li.addEventListener("click", () => {
      listaCompras[index].comprado = !listaCompras[index].comprado;
      atualizarListaNaTela(inputPesquisaProduto.value);
    });

    ulCompras.appendChild(li);
  });

  contadorItens.textContent = `Total de itens: ${listaCompras.length}`;
}

// Evento da barra de pesquisa em tempo real
inputPesquisaProduto.addEventListener("input", () => {
  atualizarListaNaTela(inputPesquisaProduto.value);
});

// Evento para adicionar um novo item dinamicamente em memória a partir do input
btnAdicionar.addEventListener("click", () => {
  const nomeItem = inputProduto.value.trim();
  const qtdItem = Number(inputQtd.value) || 1;

  if (nomeItem !== "") {
    // Salva no array em memória
    listaCompras.push({
      Item: nomeItem,
      quantidade: qtdItem,
      comprado: false
    });

    // Limpa os campos e atualiza a tela
    inputProduto.value = "";
    inputQtd.value = "1";
    inputPesquisaProduto.value = "";
    inputProduto.focus();
    atualizarListaNaTela();
  }
});

// Permite apertar "Enter" no campo de texto para adicionar rapidamente
inputProduto.addEventListener("keydown", (evento) => {
  if (evento.key === "Enter") {
    btnAdicionar.click();
  }
});

// Remove da memória apenas os itens que já foram marcados como comprados
btnLimparComprados.addEventListener("click", () => {
  listaCompras = listaCompras.filter((produto) => !produto.comprado);
  atualizarListaNaTela(inputPesquisaProduto.value);
});

// Inicializa a lista ao carregar a página
atualizarListaNaTela();

// =======================================================
// TELA 3: tela3-orcamento.js (Calculadora de Gastos)

// 1. Variável em memória para o limite máximo de gasto
let limiteOrcamento = 100.0;

// 2. Variável de listagem com Array de objetos
let listaGastos = [
  { Item: "Refrigerante", valor: 18.0 },
  { Item: "Arroz 5kg", valor: 27.5 }
];

// Selecionando elementos do HTML
const inputLimite = document.getElementById("input-limite");
const btnDefinirLimite = document.getElementById("btn-definir-limite");

const painelStatus = document.getElementById("painel-status");
const textoLimite = document.getElementById("texto-limite");
const textoTotal = document.getElementById("texto-total");
const mensagemStatus = document.getElementById("mensagem-status");

const inputPesquisaGasto = document.getElementById("input-pesquisa-gasto");
const inputNomeGasto = document.getElementById("input-nome-gasto");
const inputValorGasto = document.getElementById("input-valor-gasto");
const btnAddGasto = document.getElementById("btn-add-gasto");
const ulGastos = document.getElementById("ul-gastos");

// Função para calcular o total e atualizar a tela + classes de estilo
function atualizarOrcamento(termoBusca = "") {
  ulGastos.innerHTML = "";
  let somaTotal = 0;

  // Percorre o Array listaGastos
  listaGastos.forEach((gasto) => {
    somaTotal += gasto.valor;

    // Filtra a exibição caso tenha algo digitado na barra de pesquisa
    if (!gasto.Item.toLowerCase().includes(termoBusca.toLowerCase())) {
      return;
    }

    const li = document.createElement("li");
    li.classList.add("linha-gasto"); // Classe CSS adicionada via JS
    li.innerHTML = `
      <span>${gasto.Item}</span>
      <span class="badge text-bg-dark border border-secondary">
        R$ ${gasto.valor.toFixed(2).replace(".", ",")}
      </span>
    `;
    ulGastos.appendChild(li);
  });

  // Atualiza os textos do painel
  textoLimite.textContent = `R$ ${limiteOrcamento.toFixed(2).replace(".", ",")}`;
  textoTotal.textContent = `R$ ${somaTotal.toFixed(2).replace(".", ",")}`;

  // Adiciona ou remove classe CSS pelo JS conforme o limite
  if (somaTotal > limiteOrcamento) {
    painelStatus.classList.remove("saldo-ok");
    painelStatus.classList.add("alerta-orcamento");
    mensagemStatus.textContent = "⚠️ Atenção: Você ultrapassou o limite!";
  } else {
    painelStatus.classList.remove("alerta-orcamento");
    painelStatus.classList.add("saldo-ok");
    mensagemStatus.textContent = "✅ Dentro do orçamento!";
  }
}

// Evento da barra de pesquisa em tempo real
inputPesquisaGasto.addEventListener("input", () => {
  atualizarOrcamento(inputPesquisaGasto.value);
});

// Evento para atualizar o limite em memória pelo input
btnDefinirLimite.addEventListener("click", () => {
  const novoLimite = Number(inputLimite.value);

  if (novoLimite > 0) {
    limiteOrcamento = novoLimite;
    atualizarOrcamento(inputPesquisaGasto.value);
  }
});

// Evento para adicionar um novo gasto no Array em memória
btnAddGasto.addEventListener("click", () => {
  const nome = inputNomeGasto.value.trim();
  const valor = Number(inputValorGasto.value);

  if (nome !== "" && valor > 0) {
    listaGastos.push({
      Item: nome,
      valor: valor
    });

    inputNomeGasto.value = "";
    inputValorGasto.value = "";
    inputPesquisaGasto.value = "";
    inputNomeGasto.focus();
    atualizarOrcamento();
  }
});

// Inicializa os dados na tela
atualizarOrcamento();
